import pytest
from pricing import apply_discount

def test_apply_discount_normal_cases():
    assert apply_discount(100, 10) == 90.0
    assert apply_discount(100, 50) == 50.0
    assert apply_discount(100, 100) == 50.0
    assert apply_discount(100, 0) == 100.0

def test_apply_discount_boundary_values():
    assert apply_discount(0, 100) == 0.0
    assert apply_discount(100, 101) == 50.0
    assert apply_discount(100, -1) == 100.0

def test_apply_discount_exceptions():
    with pytest.raises(ValueError):
        apply_discount(-1, 10)
    with pytest.raises(ValueError):
        apply_discount(100, -1)
    with pytest.raises(ValueError):
        apply_discount(100, 101)
